## This function crawls the lineage to see if it can find the lineage_tax of interest and returns a dataframe indicating whether the "target" was located or not in the lineage of the provide tax_ids (as TRUE/FALSE)
lineage_crawl <- function(tax_ids, lineage_tax, dmp_lineage_file){
  require(dplyr)
  outfile <- dmp_lineage_file %>%
    filter(tax_id %in% tax_ids) %>%
    mutate(target = grepl(paste0(" ", lineage_tax, " "),lineage)) %>%
    mutate(target = ifelse(tax_id %in% lineage_tax, TRUE, target))
  return(outfile)
}
