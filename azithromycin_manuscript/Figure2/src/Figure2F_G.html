  mutate(freq = case_when(Azithromycin_binary=="Azithromycin" ~ n/93,
                          (Azithromycin_binary=="Non-Azithromycin" | Azithromycin_binary=="Other ABX") ~ (n/466)*(-1)),
         genus = factor(genus, levels=c("Bifidobacterium spp.",
                                        "Eikenella spp.", 
                                        "Fusobacterium spp.",
                                        "Haemophilus spp.", 
                                        "Klebsiella spp.", 
                                        "Megasphaera spp.",
                                        "Moraxella spp.", 
                                        "Neisseria spp.",
                                        "Proteus spp.",
                                        "Selenomonas spp.",
                                        "Staphylococcus spp.")))

keep_names <- taxa_freq_age %>% filter(abs(freq) >=0.05) %>% select(orig_name) %>% distinct()
keep_names <- taxa_freq_age %>% filter(n>1) %>% select(orig_name) %>% distinct()


taxa_freq_age <- taxa_freq_age %>% 
                           filter(orig_name %in% keep_names$orig_name) %>% 
                           mutate(orig_name=case_when(orig_name=="[Haemophilus] ducreyi" ~ "Haemophilus ducreyi",
                                                      TRUE ~ orig_name)) %>%
                           mutate(new_name = case_when(str_detect(orig_name, 'Bifidobacterium') ~ gsub("Bifidobacterium", "B.", orig_name),
                                                       str_detect(orig_name, 'Eikenella') ~ gsub("Eikenella", "E.", orig_name),
                                                       str_detect(orig_name, 'Staphylococcus') ~ gsub("Staphylococcus", "S.", orig_name),
                                                       str_detect(orig_name, 'Fusobacterium') ~ gsub("Fusobacterium", "F.", orig_name),
                                                       str_detect(orig_name, 'Megasphaera') ~ gsub("Megasphaera", "M.", orig_name),
                                                       str_detect(orig_name, 'Neisseria') ~ gsub("Neisseria", "N.", orig_name),
                                                       str_detect(orig_name, 'Haemophilus') ~ gsub("Haemophilus", "H.", orig_name),
                                                       str_detect(orig_name, 'Moraxella') ~ gsub("Moraxella", "M.", orig_name),
                                                       str_detect(orig_name, 'Proteus') ~ gsub("Proteus", "P.", orig_name),
                                                       str_detect(orig_name, 'Selenomonas') ~ gsub("Selenomonas", "S.", orig_name),
                                                       str_detect(orig_name, 'Klebsiella') ~ gsub("Klebsiella", "K.", orig_name),
                                                       TRUE ~ orig_name)) %>%
                           mutate(labels = if_else(Azithromycin_binary=="Azithromycin" & !(new_name %in% c("B. breve","F. mortiferum", "H. parainfluenzae", "M. micronuciformis", "S. ruminantium", "S. sputigena", "S. sp.", "K. sp.", "P. mirabilis")),"", paste(" ",new_name)))

microbiome_bar <- ggplot(data=taxa_freq_age,
       aes(x=reorder(new_name, -order), y=freq, group=genus, fill=genus, alpha=Azithromycin_binary)) +
