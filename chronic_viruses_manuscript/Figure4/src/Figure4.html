gamm_modeling <- function(counts, viral_load_control, binary = TRUE, event_date_cutoff){
  ## Requires a data_env be made and in the environment
  ## using the load_IMPACC_dataset function from the codebase
  model.out <- data.frame()
  plot_list <- list()
  residual_list <- list()
  
  old_comb_data <- read_csv("/scratch/metagenomics/viral_reactivation/old_full_comb_data.csv") %>% select(-1) %>%
  select(-starts_with("Human betaherpesvirus 6"),
         -starts_with("Human gammaherpesvirus 8"),
         -starts_with("Human betaherpesvirus 7"),
         -starts_with("Respiratory syncytial virus"),
         -starts_with("Human immunodeficiency virus 1"),
         -starts_with("SARS-CoV-2.pbmc"),
         -starts_with("Enterovirus.pbmc"),
         -starts_with("Enterovirus.ea"),
         -starts_with("Anelloviridae.ea"),
         -starts_with("Anelloviridae.nasal"),
         -ends_with(".ea")) %>%
  dplyr::rename("Betacoronavirus.nasal" = "SARS-CoV-2.nasal")
  
  max_part <- old_comb_data %>%
    pivot_longer(cols = -event_id) %>%
    left_join(data_env$clinical_data %>%
                distinct(event_id, participant_id, event_date)) %>%
    filter(event_date >= 0) %>%
    filter(event_date <= event_date_cutoff) %>%
    group_by(participant_id, name) %>%
    filter(name != "Betacoronavirus.nasal") %>%
    dplyr::summarize(value = max(value, na.rm = TRUE)) %>%
    filter(value != -Inf) %>%
    distinct() %>%
    pivot_wider(names_from = name, values_from = value) %>%
    ungroup() %>%
    dplyr::select(-starts_with("Betacoronavirus"))
  max_part <- max_part[complete.cases(max_part),]
  ctl_part <- max_part$participant_id[rowSums(max_part[,c(-1)]) == 0]
  
  max_part <- comb_data %>%
    pivot_longer(cols = -event_id) %>%
    left_join(data_env$clinical_data %>%
                distinct(event_id, participant_id, event_date)) %>%
    filter(event_date >= 0) %>%
    filter(event_date <= event_date_cutoff) %>%
    group_by(participant_id, name) %>%
    filter(name != "Betacoronavirus.nasal") %>%
    dplyr::summarize(value = max(value, na.rm = TRUE)) %>%
    filter(value != -Inf) %>%
    distinct() %>%
    pivot_wider(names_from = name, values_from = value) %>%
    ungroup() %>%
    dplyr::select(-starts_with("Betacoronavirus"))
  
  ## Filter for participants that we have both a nasal and pbmc sample for
  ## Really not that big of deal because we control for nasal viral load
  max_part <- max_part[complete.cases(max_part),]
  
  ## This is to standardize the control cohort to the prior version
  ## that did not yet exclude 3B plate 4 for HSV1
  
  
  for(active_vir in colnames(max_part)[-1]){
    print(paste0("Starting ", match(active_vir, colnames(max_part)[-1]), " out of ", length(colnames(max_part)[-1])))
    
    tmp.data <- counts %>%
      rownames_to_column("sample_id") %>%
      inner_join(data_env$clinical_data %>% dplyr::select(sample_id, event_id,
                                                          participant_id,
                                                          respiratory_status,
                                                          trajectory_group,
                                                          event_type,
                                                          event_date,
                                                          enrollment_site,
                                                          discretized_admit_age_quantile,
                                                          sex)) %>%
      inner_join(max_part %>% dplyr::select(participant_id, .data[[active_vir]]), by = "participant_id") %>%
      inner_join(comb_data %>%
                   dplyr::select(event_id, Betacoronavirus.nasal), by = "event_id") %>%
      relocate(sample_id, event_id, event_type, event_date,
               participant_id,
               respiratory_status, trajectory_group, enrollment_site,
               sex,discretized_admit_age_quantile, all_of(active_vir), Betacoronavirus.nasal) %>%
      filter(!is.na(.data[[active_vir]])) %>%
      mutate(sex = as.factor(sex),
             participant_id = as.factor(participant_id),
             event_date = as.numeric(event_date),
             enrollment_site = as.factor(enrollment_site),
             trajectory_group = ordered(as.factor(trajectory_group)),
             respiratory_status = ordered(as.factor(respiratory_status)),
             discretized_admit_age_quantile = ordered(as.factor(discretized_admit_age_quantile))) %>%
     filter(.data[[active_vir]] > 0 | participant_id %in% ctl_part)
    
    if(viral_load_control){
      tmp.data <- tmp.data %>%
        filter(!is.na(Betacoronavirus.nasal)) %>%
        mutate(Betacoronavirus.nasal = log2(Betacoronavirus.nasal + 1))
    }
    
    if(binary){
      ## Convert to a binary approach
      tmp.data[[active_vir]] <- ordered(as.factor(ifelse(tmp.data[[active_vir]] > 0, 1, 0)))
    } else {
      tmp.data[[active_vir]] <- log2(tmp.data[[active_vir]] +1)
    }
    
    
    ## Prepare colnames for formala/modeling
    colnames(tmp.data) <- gsub(" ", "_",
                               gsub("-", "_",
                                    gsub("\\(", "_",sub("\\)", "_",  colnames(tmp.data)))))
    cleaned_vir <- gsub(" ", "_",
                        gsub("-", "_",
                             gsub("\\(", "_",sub("\\)", "_", active_vir))))
    
   
    for(active_feature in colnames(tmp.data)[13:ncol(tmp.data)]){
      
      ## GAMM Modeling ####################
      ## Formula construction
      active_formula <- formula(paste0(active_feature, "~s(event_date, bs = 'cr') + s(event_date, bs = 'cr', by = ", cleaned_vir, ")+s(event_date, bs = 'cr', by = trajectory_group)+", cleaned_vir, "+sex+trajectory_group+discretized_admit_age_quantile"))
      
      if(viral_load_control){
        active_formula <- formula(paste0(active_feature, "~s(event_date, bs = 'cr') + s(event_date, bs = 'cr', by = ", cleaned_vir, ")+s(event_date, bs = 'cr', by = trajectory_group)+", cleaned_vir, "+sex+trajectory_group+discretized_admit_age_quantile+Betacoronavirus.nasal"))
      }
      
      tmp.data.tmp <- tmp.data %>%
        dplyr::select(event_date, enrollment_site, participant_id,
                      trajectory_group, sex, discretized_admit_age_quantile,
                      any_of(active_feature), any_of(cleaned_vir),
                      Betacoronavirus.nasal )
      tmp.data.tmp <- tmp.data.tmp[complete.cases(tmp.data.tmp),]
      
      
      tmp.data.tmp <- tmp.data.tmp %>%
        filter(event_date >= 0) %>%
        filter(event_date <= event_date_cutoff) %>%
        as.data.frame() %>%
        distinct()
    
      fit <- try(gamm4::gamm4(active_formula, random = ~(1|enrollment_site/participant_id), data = as.data.frame(tmp.data.tmp)))
      
      if(class(fit) == "try-error"){
        next
      }
      a1 =anova(fit$gam)
      ## Think about what to do here
      aa1 = sum(a1$s.table[-c(1), 3]*a1$s.table[-c(1), 2])
      aa2 = sum(a1$s.table[-c(1), 2])
      slopep=pchisq(aa1,aa2, lower.tail = F)
      slopep = a1[["s.pv"]][2]
      interceptp =a1$pTerms.pv[1]
      
      virus_n <- length(unique(tmp.data.tmp$participant_id[tmp.data.tmp[[cleaned_vir]] > 0]))
      control_n <- length(unique(tmp.data.tmp$participant_id[tmp.data.tmp[[cleaned_vir]] == 0]))

      model.out <- rbind.data.frame(model.out, c(active_vir, active_feature, slopep, unname(interceptp),
                                 sum(fit$gam$coefficients[grepl(paste0(":", cleaned_vir), names(fit$gam$coefficients))]),
                                 mean(fit$gam$coefficients[grepl(paste0(":", cleaned_vir), names(fit$gam$coefficients))]),
                                 fit$gam$coefficients[2], virus_n, control_n))
      
      if(slopep <= 0.025 | interceptp <= 0.025){
        pred <- ggpredict(fit, c("event_date", cleaned_vir), type = "fixed",
                          ci.lvl = 0.95)
        colnames(pred) <- gsub("x", "event_date", gsub("group", "virus", colnames(pred)))
        
        pred <- pred %>%
          mutate(virus = as.character(virus)) %>%
          left_join(tmp.data.tmp %>%
                      dplyr::rename("virus" = 8) %>%
                      mutate(virus = as.character(virus)) %>%
                      group_by(virus) %>%
                      dplyr::summarize(max = max(event_date)), by = "virus") %>%
          filter(event_date <= max)
        
        p1 <-pred %>%
          ggplot(mapping = aes(x = event_date, y = predicted)) +
          geom_line(mapping = aes(group = virus, y = predicted, x=event_date,
                                  color = virus), size = 0.8) +
          geom_ribbon(aes(x=event_date, ymin = conf.low, ymax = conf.high, fill = virus),
                      alpha = 0.3) +
          
          theme_classic(base_size  = 18) +
          scale_color_manual(values = c("lightgray", "#BF5700"))+
          scale_fill_manual(values = c("lightgray", "#BF5700")) +
          ylab(active_feature) +
          xlab("Days from Hospitalization") +
          ggtitle(paste0(active_vir, "\nShape = ",signif(slopep, digits = 3), "\nAverage = ", signif(interceptp, digits = 3)))
        
        plot_list <- c(plot_list, list(p1))
        names(plot_list)[length(plot_list)] <- paste0(active_feature,"___", active_vir)
        
      }
      
      if(active_vir == colnames(max_part)[2]){
        
        ## This next section is to calculate the residuals for a model WITHOUT
        ## Viral reactivation as a feature. For this reason, it only needs to be
        ## ran once, which we do during the first loop. (PS This is for the boxplot component of Figure 4)
        
        tmp.data2 <- counts %>%
          rownames_to_column("sample_id") %>%
          inner_join(data_env$clinical_data %>% dplyr::select(sample_id, event_id,
                                                              participant_id,
                                                              respiratory_status,
                                                              trajectory_group,
                                                              event_type,
                                                              event_date,
                                                              enrollment_site,
                                                              discretized_admit_age_quantile,
                                                              sex)) %>%
          inner_join(max_part %>% dplyr::select(participant_id), by = "participant_id") %>%
          inner_join(comb_data %>%
                       dplyr::select(event_id, Betacoronavirus.nasal), by = "event_id") %>%
          relocate(sample_id, event_id, event_type, event_date,
                   participant_id,
                   respiratory_status, trajectory_group, enrollment_site,
                   sex,discretized_admit_age_quantile, Betacoronavirus.nasal) %>%
          mutate(sex = as.factor(sex),
                 participant_id = as.factor(participant_id),
                 event_date = as.numeric(event_date),
                 enrollment_site = as.factor(enrollment_site),
                 trajectory_group = ordered(as.factor(trajectory_group)),
                 respiratory_status = ordered(as.factor(respiratory_status)),
                 discretized_admit_age_quantile = ordered(as.factor(discretized_admit_age_quantile)))
        
        tmp.data.tmp2 <- tmp.data2 %>%
        dplyr::select(event_date, enrollment_site, participant_id,
                      trajectory_group, sex, discretized_admit_age_quantile,
                      any_of(active_feature),
                      Betacoronavirus.nasal )
        tmp.data.tmp2 <- tmp.data.tmp2[complete.cases(tmp.data.tmp2),]
        tmp.data.tmp2 <- tmp.data.tmp2 %>%
          filter(event_date >= 0) %>%
          filter(event_date <= event_date_cutoff) %>%
          as.data.frame() %>%
          distinct()
        
        ## Construct model without viral reactivation and show residuals
        active_formula2 <- formula(paste0(active_feature, "~s(event_date, bs = 'cr') +s(event_date, bs = 'cr', by = trajectory_group) + sex+trajectory_group+discretized_admit_age_quantile"))
        if(viral_load_control){
          active_formula2 <- formula(paste0(active_feature, "~s(event_date, bs = 'cr') +s(event_date, bs = 'cr', by = trajectory_group) + sex+trajectory_group+discretized_admit_age_quantile+Betacoronavirus.nasal"))
        }
        fit2 <- try(gamm4::gamm4(active_formula2, random = ~(1|enrollment_site/participant_id), data = as.data.frame(tmp.data.tmp2)))
        
        residuals_df <- cbind.data.frame(fit2$gam$model, as.data.frame(fit2$gam$residuals)) %>%
          left_join(max_part, by = "participant_id")
        residual_list <- c(residual_list, list(residuals_df))
        names(residual_list)[length(residual_list)] <- paste0(active_feature)
      }
    }
    
  }
  return(c(list(model.out), list(plot_list), list(residual_list), list(ctl_part)))
}

traj_plot <- function(feature, viruses, results_obj, colors, log_residuals = F,
                      variable_width = 0.6, base_size = 12, better_name){
  library(ggpubr)
  if(missing(better_name)){
    feature_name <- feature
  } else {
    feature_name <- better_name
  }
  plot_list <- list()
  for(active_virus in viruses){
      plots <- results_obj[[2]][[paste0(feature, "___", active_virus)]]
      plots$data <- plots$data %>%
        mutate(virus = ifelse(virus == 0, "Control",
                              gsub("\\)", "", gsub(".*\\(", "", active_virus))))
      plots <- plots + scale_color_manual(values = colors, name = "Group")+
        scale_fill_manual(values = colors, name = "Group") +
          theme_classic(base_size  = base_size) +
        ggtitle(" ") +
          ylab(paste0("Normalized ", feature_name)) +
          theme(legend.position = "none",
                axis.text = element_text(color = "black"),
                axis.ticks = element_line(color = "black"))
      plot_list <- c(plot_list, list(plots))
  }
  
  gg2 <- as.data.frame(results_obj[[3]][[feature]]) %>%
    dplyr::rename("residuals" = 16) %>%
    pivot_longer(cols = 17:24) %>%
    mutate(value = ifelse(value > 0, 1, 0)) %>%
    group_by(participant_id, name, value) %>%
    distinct(participant_id, name, value, residuals) %>%
    slice_max(abs(residuals)) %>%
    mutate(name = ifelse(participant_id %in%results_obj[[4]], "Control", name)) %>%
    filter(value!=0|name=="Control") %>%
    mutate(name = gsub("\\)", "", gsub(".*\\(", "", name))) %>%
    filter(name %in% c("Control", gsub("\\)", "", gsub(".*\\(", "", viruses)))) %>%
    distinct() %>%
    mutate(name = fct_relevel(name, "Control")) %>%
    ggplot(aes(x=name, y=abs(residuals), color = name)) +
    #geom_violin(draw_quantiles = T) +
    geom_boxplot(outlier.shape = NA, alpha = 1, width = 0.8) +
    ggbeeswarm::geom_quasirandom(size = 0.01, alpha = 0.3) +
    theme_classic(base_size  = base_size) +
    ggtitle(feature_name) +
    scale_color_manual(values = colors) +
    theme(axis.text.x = element_blank(),
          legend.position = "none",
          axis.text = element_text(color = "black"),
            axis.ticks = element_line(color = "black"),
          axis.text.y = element_text(color = "black")) +
    ylab("Largest Magnitude Residual") +
    xlab(" ")
  if(log_residuals){
    gg2 <- gg2 + scale_y_continuous(trans='log10', labels = label_comma())
  }
      return(ggarrange(plotlist = c(list(gg2), plot_list), widths = c(variable_width,rep(1, length(plot_list))),
                       nrow = 1, align = "hv"))
}

traj_plot <- function(feature, viruses, results_obj, colors, log_residuals = F,
                      variable_width = 0.6, base_size = 12, better_name){
  library(ggpubr)
  if(missing(better_name)){
    feature_name <- feature
  } else {
    feature_name <- better_name
  }
  plot_list <- list()
  for(active_virus in viruses){
      plots <- results_obj[[2]][[paste0(feature, "___", active_virus)]]
      plots$data <- plots$data %>%
        mutate(virus = ifelse(virus == 0, "Control",
                              gsub("\\)", "", gsub(".*\\(", "", active_virus))))
      plots <- plots + scale_color_manual(values = colors, name = "Group")+
        scale_fill_manual(values = colors, name = "Group") +
          theme_classic(base_size  = base_size) +
        ggtitle(" ") +
          ylab(paste0("Normalized ", feature_name)) +
          theme(legend.position = "none",
                axis.text = element_text(color = "black"),
                axis.ticks = element_line(color = "black"))
      plot_list <- c(plot_list, list(plots))
  }
  
  gg2 <- as.data.frame(results_obj[[3]][[feature]]) %>%
    dplyr::rename("residuals" = 16) %>%
    pivot_longer(cols = 17:24) %>%
    mutate(value = ifelse(value > 0, 1, 0)) %>%
    group_by(participant_id, name, value) %>%
    distinct(participant_id, name, value, residuals) %>%
    slice_max(abs(residuals)) %>%
    mutate(name = ifelse(participant_id %in%results_obj[[4]], "Control", name)) %>%
    filter(value!=0|name=="Control") %>%
    mutate(name = gsub("\\)", "", gsub(".*\\(", "", name))) %>%
    filter(name %in% c("Control", gsub("\\)", "", gsub(".*\\(", "", viruses)))) %>%
    distinct() %>%
    mutate(name = fct_relevel(name, "Control")) %>%
    ggplot(aes(x=name, y=abs(residuals), color = name)) +
    #geom_violin(draw_quantiles = T) +
    geom_boxplot(outlier.shape = NA, alpha = 1, width = 0.8) +
    ggbeeswarm::geom_quasirandom(size = 0.01, alpha = 0.3) +
    theme_classic(base_size  = base_size) +
    ggtitle(feature_name) +
    scale_color_manual(values = colors) +
    theme(axis.text.x = element_blank(),
          legend.position = "none",
          axis.text = element_text(color = "black"),
            axis.ticks = element_line(color = "black"),
          axis.text.y = element_text(color = "black")) +
    ylab("Largest Magnitude\nResidual") +
    xlab(" ")
  if(log_residuals){
    gg2 <- gg2 + scale_y_continuous(trans='log10', labels = label_comma())
  }
      return(ggarrange(plotlist = c(list(gg2), plot_list), widths = c(variable_width,rep(1, length(plot_list))),
                       nrow = 1, align = "hv"))
}
