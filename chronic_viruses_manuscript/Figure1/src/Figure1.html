  out <- data.frame()
  for(i in colnames(matrix)){
    test_tmp <- matrix %>%
      rownames_to_column("sample_id") %>%
      pivot_longer(cols=-sample_id) %>%
      filter(!is.na(value)) 
    
    live <- test_tmp %>%
      filter(name == i) %>%
      filter(value == 1)
    
    test <- test_tmp %>%
      filter(sample_id %in% live$sample_id) %>%
      filter(value == 1) %>%
      group_by(name) %>%
      dplyr::summarize(count = n()) %>%
      mutate(freq = count / nrow(live) * 100) %>%
      dplyr::select(name, freq)
    colnames(test)[2] <- i 
     if(nrow(out) == 0){
       out <- test
     } else {
       out <- out %>%
        full_join(test)
     }
  }
  return(out)
}
