diff_exp_time_indep <- function(counts, rowfeatures, clinical_data, virus, comb_data, binary = TRUE, omic = "Same",
                     event_date_cutoff = 40, lower_date_cutoff = 0, knockin = FALSE){
  
 earliest_part <- comb_data %>%
  pivot_longer(cols = -event_id) %>%
  left_join(data_env$clinical_data %>%
              distinct(event_id, participant_id, event_date, event_type)) %>%
  filter(event_date >= lower_date_cutoff) %>%
  filter(event_date <= event_date_cutoff) %>%
  group_by(participant_id, name) %>%
  filter(name != "Betacoronavirus.nasal") %>%
  filter(value > 0) %>%
  filter(value != -Inf) %>%
  distinct() %>%
  slice_min(value, n= 1) %>%
  ungroup() %>%
  filter(!grepl(".ea|Betacorona|Simplexvirus.pbmc", name))

 
  max_part <- comb_data %>%
    pivot_longer(cols = -event_id) %>%
    left_join(data_env$clinical_data %>%
                distinct(event_id, participant_id, event_date)) %>%
    filter(event_date >= 0) %>%
    filter(event_date <= event_date_cutoff) %>%
    group_by(participant_id, name) %>%
    filter(name != "Betacoronavirus.nasal") %>%
    dplyr::summarize(value = max(value, na.rm = TRUE)) %>%
    filter(value != -Inf) %>%
    distinct() %>%
    pivot_wider(names_from = name, values_from = value) %>%
    ungroup() %>%
    dplyr::select(-ends_with(".ea"), -starts_with("Betacorona"))
  ## Filter for participants that we have both a nasal and pbmc sample for
  max_part <- max_part[complete.cases(max_part),]
  ctl_part <- max_part$participant_id[rowSums(max_part[,c(-1)]) == 0]
  
counts <- counts[,colnames(counts) %in% rownames(rowfeatures)[rowfeatures$gene_biotype == "protein_coding"]]
counts <- counts[rownames(counts) %in% clinical_data$sample_id[!is.na(clinical_data$trajectory_group)],]

event_ids_conv <- clinical_data$event_id[match(rownames(counts), clinical_data$sample_id)]
dups <- event_ids_conv[duplicated(event_ids_conv)]
counts <- counts[!rownames(counts) %in% clinical_data$sample_id[clinical_data$event_id %in% dups],]

rownames(counts) <- clinical_data$event_id[match(rownames(counts), clinical_data$sample_id)]

# Keep genes with at least 10 counts in 20% of samples
keep <- colSums(counts>=10) >= 0.2*nrow(counts)
counts <- counts[,keep]

clin.sample.event.individ.1 <- clinical_data %>%
  filter(event_id %in% rownames(counts)) %>%
  distinct(event_id, discretized_admit_age_quantile, sex, trajectory_group,
           event_type, respiratory_status) %>%
  mutate(trajectory_group = ordered(as.factor(trajectory_group)),
         respiratory_status = ordered(as.factor(respiratory_status))) %>%
  left_join(comb_data %>% dplyr::select(event_id, Betacoronavirus.nasal) %>%
              mutate(Betacoronavirus.nasal = log2(Betacoronavirus.nasal+1)))
clin.sample.event.individ.1 <- clin.sample.event.individ.1[match(rownames(counts),clin.sample.event.individ.1$event_id),]

## Filter for samples that have complete information
clin.sample.event.individ.1 <- clin.sample.event.individ.1[complete.cases(clin.sample.event.individ.1),]
counts <- counts[rownames(counts) %in% clin.sample.event.individ.1$event_id,]
clin.sample.event.individ.1 <- clin.sample.event.individ.1[clin.sample.event.individ.1$event_id %in% rownames(counts), ]

## Convert detection of the virus to yes or no
if(binary){
  if(knockin){
    virus_pos <- comb_data %>%
    distinct(event_id, .data[[virus]]) %>%
    mutate(virus_now = ifelse(is.na(.data[[virus]]) | .data[[virus]] ==0, 0, 1)) %>%
    left_join(clinical_data %>% distinct(event_id, event_date, participant_id)) %>%
      left_join(earliest_part %>% distinct(participant_id, name, event_date) %>%
                  filter(name == virus) %>%
                  dplyr::rename("min_event_date"=3)) %>%
    mutate(group = ifelse(event_date >= min_event_date, 1,NA)) %>%
      mutate(group = ifelse(participant_id %in% ctl_part, 0, group))
  } else {
    virus_pos <- comb_data %>%
    distinct(event_id, .data[[virus]]) %>%
  mutate(group = ifelse( .data[[virus]] ==0, 0, 1)) %>% ##is.na(.data[[virus]]) |
    left_join(clinical_data %>% distinct(event_id, participant_id)) %>%
    filter(group == 1 | participant_id %in% ctl_part)
  }
  
} else {
  virus_pos <- comb_data %>%
    distinct(event_id, .data[[virus]]) %>%
  mutate(group = ifelse(is.na(.data[[virus]]) | .data[[virus]] ==0, 0, log2(.data[[virus]]))) %>%
    filter(group == 1 | grepl(paste(ctl_part, collapse = "|"), event_id))
}
## New lines
tmp <- clin.sample.event.individ.1 %>% inner_join(virus_pos) %>%
  filter(!is.na(group))
counts <- counts[rownames(counts) %in% tmp$event_id,]
tmp <- tmp[tmp$event_id %in% rownames(counts), ]

metaa <- tmp %>% left_join(virus_pos)
design <- model.matrix(~ discretized_admit_age_quantile + sex + event_type + trajectory_group + Betacoronavirus.nasal + group, ## Can consider adding TG back in
                       data=metaa) 

vwts <- voom(t(counts), 
             design = design,
             normalize.method = "quantile",
             plot = T)

vfit <- lmFit(vwts)
vfit <- eBayes(vfit)

res <- topTable(vfit, coef = "group",
                sort.by = "none", number = Inf,
                adjust.method = "BH", p.value = 1)
return(res)
}
